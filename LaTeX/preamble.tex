% ── preamble.tex ──────────────────────────────────────────────────────────────
% Clean single-column, Times New Roman,
% generous margins, minimal formatting.

% ── Page layout ───────────────────────────────────────────────────────────────
\usepackage[top=1in, bottom=1in, left=1.2in, right=1.2in]{geometry}

% ── Fonts & encoding ──────────────────────────────────────────────────────────
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{times}           % Times New Roman

% ── Spacing ────────────────────────────────────────────────────────────────────
\usepackage{setspace}
\setstretch{1.1}

% ── Section headings: clean, numbered, Computer Modern bold ────────────────────
\usepackage{titlesec}
\titleformat{\section}
  {\normalfont\Large\bfseries}{\thesection}{1em}{}
\titlespacing*{\section}{0pt}{2.0ex plus 0.5ex minus 0.2ex}{1.0ex plus 0.2ex}
\titleformat{\subsection}
  {\normalfont\large\bfseries}{\thesubsection}{1em}{}
\titlespacing*{\subsection}{0pt}{1.5ex plus 0.3ex minus 0.1ex}{0.5ex}
\titleformat{\subsubsection}
  {\normalfont\normalsize\bfseries}{\thesubsubsection}{1em}{}
\titlespacing*{\subsubsection}{0pt}{1.0ex plus 0.2ex}{0.3ex}

% ── Tables & figures ──────────────────────────────────────────────────────────
\usepackage{booktabs}
\usepackage{array}
\usepackage[table]{xcolor}
\usepackage{colortbl}
\usepackage{graphicx}
\usepackage{placeins}      % \FloatBarrier to keep figures in place
\graphicspath{{figures/}}
\usepackage{caption}
\captionsetup{
  labelfont=bf,
  textfont=normalfont,
  labelsep=period,
  justification=centering,
  singlelinecheck=false,
  skip=6pt
}

% ── Math ──────────────────────────────────────────────────────────────────────
\usepackage{amsmath}
\usepackage{amssymb}

% ── Diagrams ─────────────────────────────────────────────────────────────────
\usepackage{tikz}
\usetikzlibrary{positioning, arrows.meta, decorations.pathreplacing, calc}

% ── URLs & hyperlinks ─────────────────────────────────────────────────────────
\usepackage{url}
\usepackage{hyperref}
\hypersetup{colorlinks=true, linkcolor=black, citecolor=blue!40!black, urlcolor=blue!50!black}

% ── Paragraph: indent, no extra skip ──────────────────────────────────────────
\setlength{\parindent}{1.5em}
\setlength{\parskip}{0pt}

% Citations use LaTeX's standard \cite command and keyed \bibitem entries.
% This supports multiple keys in one citation and reports unresolved references.
